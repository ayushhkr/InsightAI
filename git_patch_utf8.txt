+import pandas as pd
+
+def resolve_metric(df: pd.DataFrame, metric_name: str) -> tuple[pd.DataFrame, str]:
+    """
+    Safely resolves logical metric names into physical dataframe columns.
+    Computes derived metrics (like revenue) on-the-fly utilizing quantity, price, and discount.
+    Returns the modified dataframe and the resolved column name.
+    """
+    if not isinstance(metric_name, str):
+        return df, metric_name
+        
+    m = metric_name.lower().strip()
+    df_copy = df.copy()
+    
+    # 1. Revenue resolution
+    if m in ["revenue", "sales", "total sales", "sales revenue", "total revenue"]:
+        if "revenue" in df_copy.columns:
+            return df_copy, "revenue"
+            
+        has_qty = "quantity" in df_copy.columns
+        has_price = "price" in df_copy.columns
+        
+        disc_col = None
+        for d in ["discount", "discount_pct", "discount_rate"]:
+            if d in df_copy.columns:
+                disc_col = d
+                break
+                
+        if has_qty and has_price:
+            qty = pd.to_numeric(df_copy["quantity"], errors="coerce").fillna(0)
+            prc = pd.to_numeric(df_copy["price"], errors="coerce").fillna(0)
+            if disc_col:
+                dsc = pd.to_numeric(df_copy[disc_col], errors="coerce").fillna(0)
+                # If discounts are entered as whole percentages (e.g. 20 instead of 0.2), normalize.
+                dsc = dsc.apply(lambda x: x / 100 if x > 1 else x)
+                df_copy["revenue"] = qty * prc * (1 - dsc)
+            else:
+                df_copy["revenue"] = qty * prc
+            return df_copy, "revenue"
+        else:
+            raise ValueError("Revenue cannot be calculated because quantity and price columns are required.")
+            
+    # 2. Quantity resolution
+    if m in ["quantity", "units sold", "units", "volume"]:
+        if "quantity" in df_copy.columns:
+            return df_copy, "quantity"
+            
+    # 3. Price resolution
+    if m in ["price", "average price", "unit price"]:
+        if "price" in df_copy.columns:
+            return df_copy, "price"
+            
+    # 4. Discount resolution
+    if m in ["discount", "discount rate", "discount_pct"]:
+        for d in ["discount", "discount_pct", "discount_rate"]:
+            if d in df_copy.columns:
+                return df_copy, d
+                
+    return df_copy, metric_name
